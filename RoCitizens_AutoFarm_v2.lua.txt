-- ── Map detection ────────────────────────────────────────────────
local function IsModernMap()
    if game.PlaceId == MODERN_PLACE_ID then return true end
    for _,tag in ipairs({"ThePalms","ModernMap","PalmStreet"}) do
        if Workspace:FindFirstChild(tag) then return true end
    end
    for _,tag in ipairs({"OldMap","ClassicMap"}) do
        if Workspace:FindFirstChild(tag) then return false end
    end
    return false
end

-- ── Core helpers ─────────────────────────────────────────────────
local function SafeTP(cf)
    if not HRP then return end
    if typeof(cf)=="Vector3" then cf=CFrame.new(cf+Vector3.new(0,3,0))
    elseif typeof(cf)=="CFrame" then cf=cf+Vector3.new(0,3,0) end
    HRP.CFrame=cf; task.wait(0.4)
end
local function FindPart(names)
    local best,bestD=nil,math.huge
    for _,n in ipairs(names) do
        for _,v in ipairs(Workspace:GetDescendants()) do
            if (v:IsA("BasePart") or v:IsA("MeshPart")) and string.find(v.Name:lower(),n:lower()) then
                local d=(v.Position-HRP.Position).Magnitude
                if d<bestD then best=v bestD=d end
            end
        end
    end
    return best
end
local function Fire(name,...)
    for _,v in ipairs(ReplicatedStorage:GetDescendants()) do
        if (v:IsA("RemoteEvent") or v:IsA("RemoteFunction")) and v.Name:lower()==name:lower() then
            if v:IsA("RemoteEvent") then v:FireServer(...)
            else pcall(function() v:InvokeServer(...) end) end
            return
        end
    end
end
local function ClickPrompt()
    for _,v in ipairs(Workspace:GetDescendants()) do
        if v:IsA("ProximityPrompt") then
            local d=v.Parent:FindFirstChildWhichIsA("BasePart") or v.Parent
            if d and (d.Position-HRP.Position).Magnitude<12 then
                fireproximityprompt(v); task.wait(0.3)
            end
        end
    end
end
local function EnterPortal(pCF)
    SafeTP(pCF); task.wait(0.5)
    HRP.CFrame=pCF*CFrame.new(0,0,-2); task.wait(0.6)
    for _,r in ipairs({"StartJob","JoinJob","SelectJob","AcceptJob","EnterCareer"}) do Fire(r) end
    local vi=game:GetService("VirtualInputManager")
    if vi then
        pcall(function() vi:SendKeyEvent(true,Enum.KeyCode.F,false,game) end); task.wait(0.1)
        pcall(function() vi:SendKeyEvent(false,Enum.KeyCode.F,false,game) end)
    end
    task.wait(0.8)
end

-- ── Toggle farm ──────────────────────────────────────────────────
local function ToggleFarm(key,loopFn,pill,pillLabel,cardStroke,statusDot)
    State[key]=not State[key]
    if State[key] then
        TweenService:Create(pill,TweenInfo.new(0.18,Enum.EasingStyle.Quad),{BackgroundColor3=Color3.fromRGB(34,197,94)}):Play()
        pillLabel.Text="ON"; pillLabel.TextColor3=Color3.fromRGB(220,255,220)
        cardStroke.Color=Color3.fromRGB(34,197,94); cardStroke.Thickness=1.5
        statusDot.TextColor3=Color3.fromRGB(34,197,94)
        Loops[key]=coroutine.create(function()
            local ok,err=pcall(loopFn)
            if not ok then warn("[Hub] "..tostring(err)) end
        end)
        coroutine.resume(Loops[key])
    else
        TweenService:Create(pill,TweenInfo.new(0.18,Enum.EasingStyle.Quad),{BackgroundColor3=Color3.fromRGB(35,35,55)}):Play()
        pillLabel.Text="OFF"; pillLabel.TextColor3=Color3.fromRGB(130,130,160)
        cardStroke.Color=Color3.fromRGB(50,50,75); cardStroke.Thickness=1
        statusDot.TextColor3=Color3.fromRGB(80,80,110)
        Loops[key]=nil
    end
end
