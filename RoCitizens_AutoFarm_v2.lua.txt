local function getChar() return LP.Character end
local function getHRP()  local c=getChar(); return c and c:FindFirstChild("HumanoidRootPart") end
local function getHum()  local c=getChar(); return c and c:FindFirstChildOfClass("Humanoid") end

-- Safe teleport — waits for character, no spam
local function safeTp(cf, waitAfter)
    local hrp = getHRP()
    if not hrp then return end
    hrp.CFrame = cf
    task.wait(waitAfter or 0.6)
end

-- Walk humanoid to position and yield until arrived or timeout
local function walkTo(pos, timeout)
    local hum = getHum()
    local hrp = getHRP()
    if not hum or not hrp then return end
    hum:MoveTo(pos)
    local t = 0
    repeat
        task.wait(0.1)
        t = t + 0.1
        if t > (timeout or 8) then break end
    until (hrp.Position - pos).Magnitude < 5 or not hum
end

-- Fire all ProximityPrompts and ClickDetectors on/in a part
local function interact(part)
    if not part then return end
    for _, v in ipairs(part:GetDescendants()) do
        if v:IsA("ProximityPrompt") then
            fireproximityprompt(v)
            task.wait(0.2)
        elseif v:IsA("ClickDetector") then
            fireclickdetector(v)
            task.wait(0.2)
        end
    end
    -- also check parent chain up 2 levels
    local p = part.Parent
    for i = 1, 2 do
        if p then
            for _, v in ipairs(p:GetChildren()) do
                if v:IsA("ProximityPrompt") then fireproximityprompt(v); task.wait(0.1) end
                if v:IsA("ClickDetector")   then fireclickdetector(v);   task.wait(0.1) end
            end
            p = p.Parent
        end
    end
end

-- Find part by name list, returns {part, position}
local function findPart(names, maxDist)
    maxDist = maxDist or 999
    local hrp = getHRP()
    if not hrp then return nil end
    local best, bestDist = nil, maxDist
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Model") then
            local n = obj.Name:lower()
            for _, name in ipairs(names) do
                if n:find(name:lower()) then
                    local pos = obj:IsA("Model") and obj.PrimaryPart and obj.PrimaryPart.Position
                              or (obj:IsA("BasePart") and obj.Position)
                    if pos then
                        local d = (hrp.Position - pos).Magnitude
                        if d < bestDist then best, bestDist = obj, d end
                    end
                    break
                end
            end
        end
    end
    return best
end

local function getPos(obj)
    if not obj then return nil end
    if obj:IsA("Model") then
        return obj.PrimaryPart and obj.PrimaryPart.Position or obj:GetModelCFrame().Position
    end
    return obj.Position
end

-- Join job: TP to circle, stand in it, interact with prompt
local function joinJob(jobKey, displayName, statusFn)
    local cf = JOB_CIRCLES[jobKey]
    if not cf then return end
    statusFn("Joining " .. displayName .. " job...")
    safeTp(cf, 0.5)
    -- Find job join prompt near us
    local hrp = getHRP()
    if not hrp then return end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") or obj:IsA("ClickDetector") then
            local pos = obj.Parent and obj.Parent:IsA("BasePart") and obj.Parent.Position
                      or (obj:IsA("BasePart") and obj.Position)
            if pos and (hrp.Position - pos).Magnitude < 15 then
                if obj:IsA("ProximityPrompt") then fireproximityprompt(obj) end
                if obj:IsA("ClickDetector")   then fireclickdetector(obj)   end
                task.wait(0.3)
            end
        end
    end
    task.wait(1.5)
    statusFn(displayName .. " — Farming...")
end

-- Shared: approach a part smoothly without teleporting
local function approachAndInteract(part, token, arrivedCb)
    if not part then return end
    local pos = getPos(part)
    if not pos then return end
    local hrp = getHRP()
    if not hrp then return end
    -- Only TP if very far, otherwise walk
    local dist = (hrp.Position - pos).Magnitude
    if dist > 40 then
        safeTp(CFrame.new(pos + Vector3.new(0, 0, 3.5)), 0.5)
    else
        walkTo(pos, 6)
    end
    if arrivedCb then arrivedCb() end
    interact(part)
    task.wait(0.4)
end

-- ─── PIZZA DELIVERY (5-star optimized) ──────────────────
local function farmPizza(token, statusFn)
    -- Known pizza delivery steps: pick up order → deliver fast → return fast
    -- Speed boost during delivery for 5-star rating (time-based)
    local hum = getHum()
    if hum then hum.WalkSpeed = 28 end  -- faster walk for better rating

    while token and activeThreads["PizzaDelivery"] == token do
        statusFn("Finding order counter")
        local counter = findPart({"pizzaorder","orderstation","pickupstation","counter","pizzacounter","orderpickup"})
        if counter then
            approachAndInteract(counter, token, function() statusFn("Picking up order") end)
            task.wait(0.8)
        end

        statusFn("Delivering — 5★ rush!")
        local deliver = findPart({"deliveryspot","deliverpoint","deliver","dropoff","customer","house","pizzadrop"})
        if deliver then
            -- Sprint to delivery
            local h = getHum()
            if h then h.WalkSpeed = 36 end
            approachAndInteract(deliver, token, function() statusFn("Dropping off order") end)
            task.wait(0.5)
            if h then h.WalkSpeed = 28 end
        end

        -- Return to shop fast for next order (keeps queue and rating high)
        statusFn("Returning to shop")
        local shop = findPart({"pizzashop","pizzaplace","pizzeria","shop","restaurant"})
        if shop then
            approachAndInteract(shop, token, nil)
        end

        -- Collect tip / interact with cash register
        local register = findPart({"register","cashier","tip","collect","pay"})
        if register then
            approachAndInteract(register, token, function() statusFn("Collecting payment") end)
        end

        task.wait(1)
    end
    local h = getHum(); if h then h.WalkSpeed = 16 end
end

        statusFn("Going to fishing spot")

            approachAndInteract(sell, token, function() statusFn("Selling catch") end)

end

        local pay = findPart({"pay","register","collect","cashier","payment"})
        if pay then
            approachAndInteract(pay, token, function() statusFn("Payment received") end)
            task.wait(1)
        end
        task.wait(1)
    end
end

-- ─── CASHIER ────────────────────────────────────────────
local function farmCashier(token, statusFn)
    while token and activeThreads["Cashier"] == token do
        statusFn("Going to register")
        local reg = findPart({"register","cashier","checkout","till","scan","counter"})
        if reg then
            approachAndInteract(reg, token, function() statusFn("At register") end)
            task.wait(0.5)
        end

        statusFn("Scanning items")
        local item = findPart({"item","scan","product","barcode","grocerie","good"})
        if item then
            approachAndInteract(item, token, function() statusFn("Item scanned") end)
            task.wait(0.8)
        end

        statusFn("Processing sale")
        task.wait(1)

        local done = findPart({"complete","done","finish","sale","receipt"})
        if done then
            approachAndInteract(done, token, function() statusFn("Sale complete") end)
            task.wait(0.5)
        end
        task.wait(0.8)
    end
end
